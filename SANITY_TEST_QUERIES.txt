// Test Query for Sanity Studio - Check Auto Detail Images
// Copy and paste this into Sanity Vision (Tools > Vision in Sanity Studio)

// Query 1: Check all Service Gallery entries
*[_type == "serviceGallery"] {
  _id,
  serviceType,
  title,
  order,
  "imageUrl": image.asset->url
}

// Query 2: Check only Auto Detail entries
*[_type == "serviceGallery" && serviceType == "auto-detail"] | order(order asc) {
  _id,
  serviceType,
  title,
  order,
  "imageUrl": image.asset->url
}

// Query 3: Check all service types you've created
*[_type == "serviceGallery"] | order(serviceType asc) {
  serviceType,
  "count": count(*[_type == "serviceGallery" && serviceType == ^.serviceType])
} | unique(serviceType)

// Query 4: Check if you have any entries at all
count(*[_type == "serviceGallery"])
